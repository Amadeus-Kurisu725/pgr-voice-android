// 0.4.7（J）：测试包没有音频，每句都按字数停留——按顺序执行排定的延时任务（停留、倒计时）直到满足条件。
void Drain(Func<bool> done, int limit = 20000)
{
    for (int i = 0; i < limit && !done() && Android.OS.Handler.Delayed.Count > 0; i++)
    { var next = Android.OS.Handler.Delayed[0]; Android.OS.Handler.Delayed.RemoveAt(0); next.Action(); }
}
void Check(bool value, string message) { if (!value) throw new Exception(message); }
void Test(string name, Action action)
{
    try { action(); results.Add(new { name, passed = true }); Console.WriteLine("PASS " + name); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}

int IndexOfNode(IReadOnlyList<ListeningItem> items, string prefix) => items.ToList().FindIndex(i => i.Kind == ListeningItemKind.Line && i.NodeId.StartsWith(prefix, StringComparison.Ordinal));
ChoiceOption Option(Pack pack, string tail) => pack.Nodes.SelectMany(n => n.Options).Single(o => o.Id == P(tail));

Test("听书核心 Manual：选择后同样续接，到下一菜单停下等待选择而非等待续接", () =>
{
    var pack = Load(); var s = new CoreSession(pack, ChapterOf(pack), ListeningBranchPolicy.Manual);
    int menus = 0, guard = 0; bool passedStall = false;
    while (s.Current is { } item && item.SectionId != Section && guard++ < 5000) s.MoveNext();
    while (s.Current is { } item && item.SectionId == Section && guard++ < 5000)
    {
        Check(!item.IsBlocking, "手动选择仍停在等待续接：" + item.Notice);
        if (item.Kind == ListeningItemKind.Choice) { menus++; Check(s.Choose(item.Options![0].Id), "无法选择"); continue; }

foreach (int number in new[] { 1, 2 })
Test($"游戏伴音：默认分支 {number} 号自动推进走完 38-1（含嵌套饮品菜单），每个段尾都按登记连接续接", () =>
{
    var pack = Load(); var e = new PlaybackEngine(pack);
    Check(e.OpenGameMenu(P("menu-000"), Section), "无法定位");
    int guard = 0, continuations = 0, manualInteractions = 0; var heard = new List<string>();
    while (guard++ < 2000 && !(e.Current?.Id.StartsWith(Common, StringComparison.Ordinal) ?? false))
    {
        if (e.Mode == RunMode.Choice)
        {
            var menu = e.Current!; e.Commit(menu.Id);
            if (DefaultBranchPolicy.TryChoice(e, number, out var o, out _)) e.SelectBranch(e.AvailableOptions.FindIndex(x => x.Id == o!.Id));
            else
            {
                // 正文未核的选项（menu-006/008 只有地点/操作标题）：自动播放出示确认卡，玩家确认后导航器拒绝进入（ReviewRoute），
                // 现稿续接按包内写的目标继续。
                var only = menu.Options[DefaultBranchPolicy.EffectiveNumber(menu, number) - 1]; Check(!only.BodyVerified, "意外的需确认菜单 " + menu.Id);
                e.SelectBranch(e.AvailableOptions.FindIndex(x => x.Id == only.Id));
                Check(e.Mode == RunMode.Choice && e.ReviewRoute == only.Id, "导航器未按原样拒绝未核正文");
                Check(DeclaredContinuationPolicy.TryResolve(e, out var dc) && DeclaredContinuationPolicy.Follow(e, dc!), "未核正文没有按现稿续接");
                manualInteractions++;
            }
            continue;
        }
        if (e.Mode == RunMode.Following && e.Current?.Kind == "line") heard.Add(e.CurrentId!);
        if (DeclaredContinuationPolicy.TryResolve(e, out var c)) { continuations++; Check(DeclaredContinuationPolicy.Follow(e, c!), "续接失败"); continue; }
        if (e.Mode is RunMode.Following or RunMode.Merge) { e.Next(true); continue; }
        throw new Exception($"停在 {e.Mode} {e.CurrentId} {e.Notice}");
    }
    Check(e.Current?.Id.StartsWith(Common, StringComparison.Ordinal) == true, $"没有到达共同台词：{e.Mode} {e.CurrentId} guard={guard} heard={heard.Count} cont={continuations} 最后={string.Join(",", heard.TakeLast(3))}");
    Check(manualInteractions == 2, "未核互动次数异常 " + manualInteractions);
    Check(heard.Contains(Stall) && continuations >= 6, $"续接次数异常 {continuations}");
    var drink = Option(pack, number == 2 ? "route-002-01" : "route-002-00");
    Check(drink.LineIds.All(heard.Contains), "饮品选项与设定序号不符");
    Console.WriteLine($"   默认 {number}：台词 {heard.Count} 句，登记续接 {continuations} 次，未核正文互动按现稿续接 {manualInteractions} 处");
});

// ---- 0.4.7 游戏伴音终点规则：未归档 end 节点（选项出口 / 小节正文末尾）不关闭自动播放，提示终点文字，等待游戏画面定位，定位到下一句自动续播 ----
// 与 AppSession.AdvanceAfterAutoLine / TryBeginGameEndWait / HandleGameEndWaitObservation 相同的判定顺序：
// 本句播完（Tapping）→ 续播被拒 → GameEndWaitPolicy.TryBegin → AutoPlaybackCycle.BeginWaiting（仍 Running、无超时）
// → 读到的画面用 Matcher.FindAll 在终点小节与下一小节里找唯一台词 → CanResume（导航器确认）→ Start/PreparePlayback 续播。
List<OcrBlock> Screen(Node node) => new() { new OcrBlock { Text = node.Text, Score = .99, Box = new[] { new double[] { 0, 0 }, new double[] { 400, 0 }, new double[] { 400, 40 }, new double[] { 0, 40 } } } };
(GameEndWait Wait, AutoPlaybackCycle Cycle) EndWaitAt(PlaybackEngine e, string label)
{
    var cycle = new AutoPlaybackCycle(); long now = 1000;
    cycle.Start(e.CurrentId!, now);
    Check(cycle.PreparePlayback(cycle.Epoch, e.CurrentId!, now) && cycle.BindAudio(cycle.Epoch, e.CurrentId!, 7, now) && cycle.CompleteAudio(cycle.Epoch, 7, now + 3000), label + "：末句没有播完");
    var decision = e.EvaluateCommonAutoPlayNext();
    Check(!(decision.Allowed && decision.Next != null), label + "：终点前竟允许点击下一句 " + decision.Next?.Id);
    Check(!decision.Reason.Contains("共同剧情已结束") && !decision.Reason.Contains("本段已播放完毕"), label + "：仍显示旧的结束提示：" + decision.Reason);
    Check(GameEndWaitPolicy.TryBegin(e, decision, out var wait) && wait != null, label + "：没有进入终点等待（" + decision.Code + "）");
    Check(cycle.BeginWaiting(cycle.Epoch, wait!.SourceId), label + "：自动播放没有转入等待");
    Check(cycle.Running && cycle.WaitingForGame && !cycle.TimedOut(now + 3_600_000), label + "：等待中自动播放被关闭或超时");
    Check(!cycle.BeginTap(cycle.Epoch, "x", now + 4000), label + "：等待中仍会点击游戏");
    PlaybackLog.Write(PlaybackLog.Types.EndWait, "自动播放", e.Pack, "", wait.SectionId, wait.EndId ?? wait.SourceId, GameEndWaitPolicy.LogReason(wait));
    return (wait, cycle);
}
Node? Located(PlaybackEngine e, GameEndWait wait, Node screen)
{
    var blocks = Screen(screen);
    var exact = wait.SearchSections.SelectMany(sec => Matcher.FindAll(e, sec, blocks))
        .Where(c => c.Node.Kind == "line" && c.Node.Id != wait.SourceId && Matcher.Normalize(c.Evidence) == Matcher.Normalize(c.Node.Text))
        .GroupBy(c => c.Node.Id).Select(g => g.First().Node).ToList();
    return GameEndWaitPolicy.Pick(e.Pack, wait, exact);
}
void ResumeAt(PlaybackEngine e, GameEndWait wait, AutoPlaybackCycle cycle, Node next, string label)
{
    var source = e.Pack.ById[wait.SourceId];
    Check(Located(e, wait, source) == null, label + "：画面仍是末句时被当成新位置");
    Check(!GameEndWaitPolicy.CanResume(e, wait, source, out _, out _), label + "：末句本身结束了等待");
    Check(cycle.WaitingForGame, label + "：末句画面让等待结束");
    var found = Located(e, wait, next);
    Check(found?.Id == next.Id, label + "：画面出现下一句却没有唯一定位到 " + next.Id + "（" + found?.Id + "）");
    Check(GameEndWaitPolicy.CanResume(e, wait, found!, out var probe, out var why) && probe != null, label + "：定位到下一句仍不能续播：" + why);
    cycle.Start(found!.Id, 9000);
    Check(cycle.PreparePlayback(cycle.Epoch, found.Id, 9000) && cycle.Phase == AutoPlaybackPhase.Playing, label + "：没有从定位句恢复播放");
    Check(e.ConfirmGameLine(found.Id) && e.CurrentId == found.Id && e.Mode == RunMode.Following, label + "：导航器没有确认定位句");
}
Node FirstLine(Pack pack, string sectionId)
{
    var start = pack.Chapters.SelectMany(c => c.Sections).First(x => x.Id == sectionId).StartId;
    var node = pack.ById[start];
    for (int i = 0; i < 50 && node.Kind != "line" && node.NextId != null; i++) node = pack.ById[node.NextId];
    return node;
}

(List<string> Lines, int Stops) QuizListen(Pack pack, string sec, ListeningBranchPolicy policy, int number)
{
    var core = new CoreSession(pack, ChapterFor(pack, sec), policy); int stops = 0, guard = 0;
    while (core.Current is { } item && guard++ < 50000)
    {
        Check(!(item.IsBlocking && item.WaitingKind == ListeningWaitingKind.TraversalLimit), policy + " 过深停下：" + item.Notice);
        if (item.Kind == ListeningItemKind.Choice || item.IsBlocking)
        {
            if (item.IsBlocking && item.SectionId == sec) stops++;
            if (core.ResolveStop(number) is { } st && core.ApplyStop(st)) continue;
            if (item.SectionId == sec) stops++; break;
        }
        core.MoveNext();
    }
    return (core.Items.Where(i => i.SectionId == sec && i.Kind == ListeningItemKind.Line).Select(i => i.NodeId).ToList(), stops);
}
// 游戏伴音：从小节开头自动播放；菜单按 DefaultBranchPolicy（单选项 / 同回应隐藏后单选项 / 默认分支 / 探索顺序）自动选入，与 AppSession 相同记入探索记忆。
(List<string> Played, int Handovers, int HiddenSingles, List<string> Chosen) QuizGame(Pack pack, string sec, int number)
{
    var start = pack.Chapters.SelectMany(c => c.Sections).First(x => x.Id == sec).StartId;
    var e = new PlaybackEngine(pack); Check(e.ConfirmGameLine(start) || e.OpenGameMenu(start, sec), "无法定位小节开头");
    var played = new List<string>(); var chosen = new List<string>(); int handovers = 0, hiddenSingles = 0;
    for (int guard = 0; guard < 20000 && e.Mode != RunMode.End && e.Current?.SectionId == sec; guard++)
    {
        if (e.Mode == RunMode.Choice)
        {
            var menuId = e.CurrentId!; e.Commit(menuId);
            bool single = DefaultBranchPolicy.IsSingleOptionMenu(e);
            if (DefaultBranchPolicy.IsSingleAfterHiding(e)) hiddenSingles++;
            if (!DefaultBranchPolicy.TryChoice(e, single ? 1 : number, out var o, out var first)) { handovers++; break; }
            chosen.Add(o!.Label); ExplorationMemory.Add(e, menuId, o.Id);
            var sh = MenuShape.Of(pack, pack.ById[menuId]);
            if (sh.Exploration) foreach (var em in sh.EmptyLoops) ExplorationMemory.Add(e, menuId, em.Id);
            Check(DefaultBranchPolicy.Enter(e, o, first!), "选入失败 " + menuId);
            continue;
        }
        if (e.Mode == RunMode.Following && e.Current is { Kind: "line" } line) played.Add(line.Id);
        if (e.Mode is RunMode.Following or RunMode.Merge)
        {
            if (DeclaredContinuationPolicy.TryResolve(e, out var c) && c != null && DeclaredContinuationPolicy.Follow(e, c)) continue;
            if (e.Current?.NextId == null && e.Current?.Kind == "line") break;
            e.Next(true); continue;
        }
        handovers++; break;
    }
    return (played, handovers, hiddenSingles, chosen);
}
