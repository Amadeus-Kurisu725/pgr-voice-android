namespace Android.OS
{
    public interface IBinder { }
    public static class Looper { public static object MainLooper { get; } = new(); }
    public static class SystemClock { public static long Now; public static long ElapsedRealtime() => Now; }
    public sealed class Handler(object looper)
    {
        static readonly List<(Handler Owner, Action Action, long At)> queue = [];
        public void Post(Action action) => action();
        public void PostDelayed(Action action, long delay) => queue.Add((this, action, SystemClock.Now + delay));
        public void RemoveCallbacks(Action action) => queue.RemoveAll(x => x.Owner == this && x.Action == action);
        public void RemoveCallbacksAndMessages(object? token) => queue.RemoveAll(x => x.Owner == this);
        public static void Reset() { queue.Clear(); SystemClock.Now = 0; }
        public static Action? IdleCallback => queue.FirstOrDefault(x => x.At - SystemClock.Now >= 299_000).Action;
        public static void Advance(long milliseconds)
        {
            long target = SystemClock.Now + milliseconds;
            for (int i = 0; i < 100_000; i++)
            {
                var ready = queue.Where(x => x.At <= target).OrderBy(x => x.At).FirstOrDefault();
                if (ready.Action == null) { SystemClock.Now = target; return; }
                queue.Remove(ready); SystemClock.Now = ready.At; ready.Action();
            }
            throw new Exception("计时夹具队列未收敛");
        }
    }
    public enum WakeLockFlags { Partial }
    public sealed class PowerManager
    {
        public WakeLock? Latest;
        public WakeLock NewWakeLock(WakeLockFlags flags, string tag) => Latest = new();
        public sealed class WakeLock : IDisposable
        {
            public bool IsHeld { get; private set; }
            public void SetReferenceCounted(bool value) { }
            public void Acquire() => IsHeld = true;
            public void Release() => IsHeld = false;
            public void Dispose() => Release();
        }
    }
